// Generates KobanCore/Features/Settings/L10nTables.swift from src/i18n/locales/*.ts
// Usage: node --experimental-strip-types gen-l10n.mjs <koban repo> <out.swift>
import { readdirSync, writeFileSync, readFileSync } from 'node:fs';
import { pathToFileURL } from 'node:url';
import path from 'node:path';
const [repo, out] = process.argv.slice(2);
const dir = path.join(repo, 'src/i18n/locales');
const files = readdirSync(dir).filter(f => f.endsWith('.ts')).sort();
// LANGUAGES table from i18n/index.tsx
const idx = readFileSync(path.join(repo, 'src/i18n/index.tsx'), 'utf8');
const langRe = /\{ code: '([^']+)', native: '([^']+)', english: '([^']+)', intl: '([^']+)', group: '([^']+)' \}/g;
const langs = [...idx.matchAll(langRe)].map(m => m.slice(1));
const esc = s => JSON.stringify(s).replace(/\\u([0-9a-fA-F]{4})/g, (_, h) => `\\u{${h}}`).replace(/\$/g, '$');
let swift = `// GENERATED by tools/gen-l10n.mjs from src/i18n/locales/*.ts — do not edit by hand.\n// Re-run: node --experimental-strip-types gen-l10n.mjs <repo> <this file>\n\nenum L10nTables {\n`;
swift += `    /// (code, native, english, intl, group) — LANGUAGES in src/i18n/index.tsx, same order.\n`;
swift += `    static let languages: [(code: String, native: String, english: String, intl: String, group: String)] = [\n`;
for (const l of langs) swift += `        (${l.map(esc).join(', ')}),\n`;
swift += `    ]\n\n`;
const names = [];
for (const f of files) {
  const code = f.replace(/\.ts$/, '');
  const mod = await import(pathToFileURL(path.join(dir, f)).href);
  const dict = Object.values(mod).find(v => v && typeof v === 'object');
  const ident = 'lang_' + code.replace(/-/g, '_');
  names.push([code, ident]);
  swift += `    static let ${ident}: [String: String] = [\n`;
  for (const [k, v] of Object.entries(dict)) swift += `        ${esc(k)}: ${esc(v)},\n`;
  swift += `    ]\n\n`;
}
swift += `    static let tables: [String: [String: String]] = [\n`;
for (const [c, i] of names) swift += `        ${esc(c)}: ${i},\n`;
swift += `    ]\n}\n`;
writeFileSync(out, swift);
console.log(`wrote ${names.length} locales, ${langs.length} languages`);
